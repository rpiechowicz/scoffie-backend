// -----------------------------
// SAFE DAILY COMMANDS (NO DB RESET)
// -----------------------------

// Start/restart stack (keeps DB volume and data)
docker compose up -d --build
docker compose stop
docker compose down --remove-orphans

// Apply existing migrations safely (never use migrate dev on persistent DB)
// WINDOWS
docker compose exec api --% pnpm prisma migrate deploy
// MACOS
docker compose exec api pnpm prisma migrate deploy

// Optional seed (may add/overwrite seed records depending on seed script logic)
// WINDOWS
docker compose exec api --% pnpm prisma:seed
// MACOS
docker compose exec api pnpm prisma:seed

docker compose exec api pnpm prisma studio -- --hostname 0.0.0.0 --port 5555

// WEBSOCKET SMOKE TESTS (Socket.IO)
docker compose exec api pnpm ws:smoke users:findAll '{}'
docker compose exec api pnpm ws:smoke households:findAll '{"userId":"<USER_ID>"}'
docker compose exec api pnpm ws:smoke recipes:findAll '{"userId":"<USER_ID>"}'
docker compose exec api pnpm ws:smoke weeklyPlans:listByHousehold '{"userId":"<USER_ID>","householdId":"<HOUSEHOLD_ID>"}'

// TRANSLATE EXISTING RECIPES TO POLISH
docker compose exec api pnpm run translate:recipes:pl

// GENERATE RECIPE IMAGES (fills imageUrl)
docker compose exec api pnpm run generate:recipe:images

// UPLOAD LOCAL RECIPE IMAGES TO S3 + UPDATE imageUrl
docker compose exec api pnpm run upload:recipe:images:s3

// Import recipes safely (do NOT clear existing)
docker compose exec api sh -lc "IMPORT_CLEAR_EXISTING=false pnpm run import:recipes"
